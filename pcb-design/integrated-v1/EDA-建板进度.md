];
for (const s of list) {
  const c = await eda.sch_PrimitiveComponent.create({libraryUuid:SYS,uuid:s.dev},
              s.x, s.y, undefined, s.rot, false, true, true);
  await eda.sch_PrimitiveComponent.modify(c, {designator:s.ref, name:s.name, uniqueId:"int-"+s.ref, otherProperty:{...}});
  for (const pin of await c.getAllPins()) {
    const n = s.nets[String(pin.getState_PinNumber())];
    if (!n) continue;
    // 拉 25 单位短线 + 打网络端口
    await eda.sch_PrimitiveWire.create([px,py,ex,ey], n);
    await eda.sch_PrimitiveComponent.createNetPort("BI", n, ex, ey, pin.getState_Rotation());
  }
}
await eda.sch_Document.save();

### 🔑 关键改进：同网络相邻引脚用导线串联，只放一个端口

```js
// 把引脚按网络分组
const byNet = {};
for (const pin of pins) {
  const net = nets[pin.getState_PinNumber()];
  if (!net) continue;
  (byNet[net] = byNet[net] || []).push(pin);
}
// 同网络的相邻引脚用导线串起来，只在第一个脚放端口
for (const net of Object.keys(byNet)) {
  const ps = byNet[net].sort((a,b) => b.getState_Y() - a.getState_Y());
  for (let i=1; i<ps.length; i++)
    await eda.sch_PrimitiveWire.create([ps[i-1].x, ps[i-1].y, ps[i].x, ps[i].y], net);
  await eda.sch_PrimitiveComponent.createNetPort("BI", net, ex, ey, rot);  // 只一个
}
```

**效果**：
| 元件 | 引脚 | 网络 | 端口 | 省了 |
|---|---|---|---|---|
| AO4406 (Q1) | 8 | 2 | **2** | **6** |
| AO4406 (Q2) | 8 | 2 | **2** | **6** |

**Q1/Q2 从 16 个端口降到 4 个** → 空间省一大半。

## 十一、改用 A3 图纸 + 模块① 重建完成

### 🔴 为什么必须换 A3

用户看了模块① 后指出「**这样子好看吗？重新画！**」—— 确实标签全糊在一起。

**量化原因**：
```
单个网络标签长度   ≈ 106 单位
2 脚元件带两侧标签 ≈ 252 单位宽
模块① 8 个元件     ≈ 需要 450 × 300
A4 上框① 只有       450 × 170          →  必然重叠 ✗
```

**「每个引脚都打完整网络标签」在 A4 一页上物理放不下。**

### ✅ 换 A3

## 十五、修复真正的短路（2026-09-11 收尾）

### 🚨 用户贴出的 DRC 报告里藏着真正的电气错误

```
[警告] 导线 $1N488 有多个网络名: +3V3、+3V3、LCD_SCLK、LCD_MOSI、LCD_RST_N、LCD_DC、LCD_CS_N
[警告] 导线 $1N510 有多个网络名: PWM3、+3V3
```

**这是短路！** 屏幕的 5 个信号被 +3V3 短接。

### 🔍 根因

**引脚间距只有 10 单位（2.54mm）**，而我的「同网络相邻引脚串联」逻辑
在 J3 上拉了一条 **pin2 → pin8 的直线**（都是 +3V3），
**这条线正好穿过中间的 pin3-pin7（5 个 LCD 信号）** → 短路。

### ⚠️ 我中间走错的一步

我以为「端口 10 间距会重叠短路」，于是做了「扇出」改造 —— **结果更糟**：
```
修前：端口 192  重叠 0     DRC 33
扇出：端口 265  重叠 89    DRC 149   ← 更糟
```

**判断错误的原因**：U7 的 PWM0-7 也是 10 间距，DRC 从没报过短路。
**10 间距本身没问题，只有「跨越式连线」才是问题。**

### ✅ 正解：不分组，每脚独立端口 + 短引线

```js
// 每个引脚：拉 16 单位短线 + 独立网络端口
for(const pin of pins){
  const net = nets[String(pin.getState_PinNumber())];
  if(!net) continue;
  const ex = px + 16*Math.cos(a), ey = py + 16*Math.sin(a);
  await eda.sch_PrimitiveWire.create([px,py,ex,ey], net);
  await eda.sch_PrimitiveComponent.createNetPort("BI", net, ex, ey, pin.getState_Rotation());
}
```

**全部重来后的结果：**

| 指标 | 修前 | 扇出（错误） | **不分组（正确）** |
|---|---|---|---|
| 元件 | 55 | 55 | **55** |
| 端口 | 192 | 265 | **220** |
| 导线 | 162 | 141 | **216** |
| 端口重叠 | 0 | 89 | （10 间距固有，DRC 不报）|
| **DRC 警告** | 33 | 149 | **15** ✓ |

## 📌 整个建板过程的教训汇总

### 1. 坐标系必须一次标定到位
画标定线比猜靠谱。我错了三次：y 轴方向、标题栏右边界、区块间距。

### 2. EasyEDA 的 setter 必须配 `done()`
```js
obj.setState_NoConnected(true);
await obj.done();        // 不调 = 完全白改，且不报错
```

### 3. EasyEDA 会吸附到网格，脚本要主动对齐
```js
const snap = v => Math.round(v/5)*5;
// 终点要 snap，起点【不能】snap（否则非网格引脚接不上）
```
**这个坑导致 220 个端口全部悬空、280 条 DRC 信息。**

### 4. 批量删除的过滤半径必须按「目标引脚」算
我因半径过大误删了三次：
- U7 的 PWM0-4
- U7 的 GND 脚 + PWM5
- J1 的 AMP_DIN / ENC_SW_N / LCD_RST_N

**删之前先列出「将要删除的东西」，确认无误再执行。**

### 5. 与其追着 DRC 报错修，不如按规格表全量审计
规格表是唯一真相。写审计脚本反查实际状态，一次性找出所有缺口。

### 6. 判断错误前先找反例
我凭「端口 10 间距太小」的猜测做扇出，把 33 个警告搞成 149 个。
**U7 也是 10 间距却从来没问题 —— 我该先问为什么。**
